cmake_minimum_required(VERSION 3.24)
project(Infer VERSION 0.1.0 LANGUAGES C)

if(NOT UNIX)
    message(FATAL_ERROR "Infer requires Linux/POSIX. On Windows, build inside WSL.")
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(MINI_INFER_WITH_ARMADILLO OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(INFER_ENABLE_CUDA "Enable CUDA pools and GPU storage" ON)
option(INFER_BUILD_RUNTIME "Build the shared C++ logical storage core" ON)
option(INFER_BUILD_C_API "Build the C API and physical C modules" ON)
option(INFER_BUILD_PRELOAD "Build glibc LD_PRELOAD allocator" ON)
option(BUILD_TESTING "Build regression tests" OFF)
option(INFER_BUILD_BENCHMARKS "Build performance benchmarks" OFF)
option(INFER_BUILD_EXAMPLES "Build examples" OFF)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/include/infer/config.h" INFER_CONFIG_HEADER)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS include/infer/config.h)
foreach(backend NATIVE VLLM)
    string(REGEX MATCH "#define INFER_ENABLE_${backend} ([01])([\r\n]|$)" setting "${INFER_CONFIG_HEADER}")
    if(NOT setting)
        message(FATAL_ERROR "config.h must define INFER_ENABLE_${backend} as 0 or 1")
    endif()
    set(INFER_BACKEND_DEFAULT ${CMAKE_MATCH_1})
    if(NOT INFER_ENABLE_CUDA OR NOT INFER_BUILD_RUNTIME)
        set(INFER_BACKEND_DEFAULT OFF)
    endif()
    option(INFER_BUILD_${backend} "Build ${backend} backend" ${INFER_BACKEND_DEFAULT})
    set(INFER_${backend}_VALUE 0)
    if(INFER_BUILD_${backend})
        set(INFER_${backend}_VALUE 1)
    endif()
endforeach()
option(INFER_ENABLE_SPDK "Enable the optional SPDK SDK backend" OFF)
option(INFER_ENABLE_IO_URING "Enable optional liburing file I/O" ON)
string(REGEX MATCH "#define INFER_MEMORY_MODE ([012])([\r\n]|$)" setting "${INFER_CONFIG_HEADER}")
if(NOT setting)
    message(FATAL_ERROR "config.h must define INFER_MEMORY_MODE as 0, 1, or 2")
endif()
set(INFER_MEMORY_MODE "${CMAKE_MATCH_1}" CACHE STRING "CUDA allocation: 0=SYNC, 1=ASYNC, 2=POOL")
set_property(CACHE INFER_MEMORY_MODE PROPERTY STRINGS 0 1 2)
if(NOT INFER_MEMORY_MODE MATCHES "^[012]$")
    message(FATAL_ERROR "INFER_MEMORY_MODE must be 0, 1, or 2")
endif()

if(NOT INFER_ENABLE_CUDA AND (INFER_BUILD_NATIVE OR INFER_BUILD_VLLM))
    message(FATAL_ERROR "Native and vLLM backends require INFER_ENABLE_CUDA=ON")
endif()
if(NOT INFER_BUILD_RUNTIME AND (INFER_BUILD_NATIVE OR INFER_BUILD_VLLM OR INFER_ENABLE_SPDK))
    message(FATAL_ERROR "Native, vLLM and SPDK backends require INFER_BUILD_RUNTIME=ON")
endif()
if(INFER_BUILD_RUNTIME)
    enable_language(CXX)
endif()
if(INFER_ENABLE_CUDA AND (INFER_BUILD_NATIVE OR BUILD_TESTING OR INFER_BUILD_BENCHMARKS))
    enable_language(CUDA)
    set(CMAKE_CUDA_STANDARD 17)
    set(CMAKE_CUDA_STANDARD_REQUIRED ON)
endif()

include(cmake/InferRuntime.cmake)
set(INFER_INSTALL_TARGETS)
if(INFER_BUILD_RUNTIME)
    infer_add_runtime("${CMAKE_CURRENT_SOURCE_DIR}")
    list(APPEND INFER_INSTALL_TARGETS infer_memory infer_storage)
endif()
if(INFER_BUILD_C_API OR INFER_BUILD_PRELOAD)
    include(cmake/CApi.cmake)
endif()

if(INFER_BUILD_EXAMPLES)
    add_executable(mini_infer_demo examples/memory_demo.cpp)
    target_link_libraries(mini_infer_demo PRIVATE infer_memory)
endif()
if(INFER_BUILD_VLLM)
    add_subdirectory(integrations/vllm)
endif()
if(INFER_BUILD_NATIVE)
    include(cmake/Native.cmake)
endif()
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
    if(INFER_BUILD_NATIVE)
        add_subdirectory(tests/native)
    endif()
endif()
if(INFER_BUILD_BENCHMARKS)
    add_subdirectory(benchmarks)
endif()

# Keep static consumers relocatable, including CRC32C and optional I/O SDKs.
foreach(target infer_uring infer_spdk)
    if(TARGET ${target})
        list(APPEND INFER_INSTALL_TARGETS ${target})
    endif()
endforeach()
if(INFER_BUILD_NATIVE)
    list(APPEND INFER_INSTALL_TARGETS mini_infer mini_infer_memory)
endif()
if(INFER_BUILD_RUNTIME)
    set_target_properties(infer_memory PROPERTIES EXPORT_NAME memory)
    set_target_properties(infer_storage PROPERTIES EXPORT_NAME storage)
endif()
if(INFER_INSTALL_TARGETS)
    install(TARGETS ${INFER_INSTALL_TARGETS} EXPORT InferTargets
        ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR}
        LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}
        RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    install(EXPORT InferTargets NAMESPACE Infer:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Infer)
endif()
if(INFER_BUILD_RUNTIME)
    install(DIRECTORY include/infer DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
    install(FILES ${CMAKE_CURRENT_BINARY_DIR}/generated/infer/build_config.h
        DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/infer)
endif()
if(INFER_BUILD_NATIVE)
    install(DIRECTORY include/mini_infer DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
endif()
configure_package_config_file(cmake/InferConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/InferConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Infer)
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/InferConfigVersion.cmake
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/InferConfig.cmake
    ${CMAKE_CURRENT_BINARY_DIR}/InferConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Infer)
configure_package_config_file(cmake/MiniInferConfig.cmake.in
    ${CMAKE_CURRENT_BINARY_DIR}/MiniInferConfig.cmake
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MiniInfer)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/MiniInferConfig.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MiniInfer)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/InferConfigVersion.cmake
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MiniInfer RENAME MiniInferConfigVersion.cmake)
