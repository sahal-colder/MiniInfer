cmake_minimum_required(VERSION 3.24)
project(infer VERSION 3.0.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(INFER_WERROR "Treat warnings as errors" ON)
option(INFER_BUILD_RUNTIME "Build shared logical core and C business API" ON)
option(INFER_BUILD_PRELOAD "Build glibc LD_PRELOAD allocator" ON)
option(INFER_ENABLE_CUDA "Enable CUDA physical pools" OFF)
option(INFER_IO_DLSYM "Build RTLD_NEXT file adapter" ON)
option(INFER_IO_URING "Build optional liburing file adapter when available" ON)
option(INFER_BENCH "Build benchmarks" OFF)
option(BUILD_TESTING "Build regression tests" OFF)
add_compile_options(-Wall -Wextra)
if(INFER_WERROR)
  add_compile_options(-Werror)
endif()
include(CTest)
include(GNUInstallDirs)
find_package(Threads REQUIRED)

add_library(infer_base STATIC src/base/crc32c.c)
target_include_directories(infer_base PUBLIC include)
add_library(infer_mem STATIC src/mem/mem.c)
target_link_libraries(infer_mem PUBLIC infer_base Threads::Threads)
add_library(infer_io STATIC src/io/io.c src/io/posix.c)
target_link_libraries(infer_io PUBLIC infer_base)
if(INFER_IO_DLSYM)
  target_sources(infer_io PRIVATE src/io/dlsym.c)
  target_compile_definitions(infer_io PRIVATE IF_HAVE_DLSYM=1)
  target_link_libraries(infer_io PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
endif()
if(INFER_IO_URING)
  find_package(PkgConfig QUIET)
  if(PkgConfig_FOUND)
    pkg_check_modules(V2_URING QUIET IMPORTED_TARGET liburing)
  endif()
  if(TARGET PkgConfig::V2_URING)
    target_sources(infer_io PRIVATE src/io/uring.c)
    target_compile_definitions(infer_io PRIVATE IF_HAVE_URING=1)
    target_link_libraries(infer_io PRIVATE PkgConfig::V2_URING Threads::Threads)
  else()
    message(STATUS "liburing unavailable: explicit IF_IO_URING requests return -ENOTSUP")
  endif()
endif()
foreach(layer infer_base infer_mem infer_io)
  target_compile_options(${layer} PRIVATE -fvisibility=hidden)
endforeach()
if(INFER_BUILD_RUNTIME)
  enable_language(CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  get_filename_component(INFER_SOURCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
  include(${INFER_SOURCE_ROOT}/cmake/InferRuntime.cmake)
  infer_add_runtime(${INFER_SOURCE_ROOT})
  foreach(layer infer_memory infer_storage)
    target_compile_options(${layer} PRIVATE -fvisibility=hidden)
  endforeach()
  add_library(infer_store STATIC src/store/store.cpp src/runtime/runtime.cpp src/runtime/stream.cpp)
  target_link_libraries(infer_store PUBLIC infer_mem infer_io infer_storage)
  target_compile_options(infer_store PRIVATE -fvisibility=hidden)
  add_library(infer_aio STATIC src/aio/aio.c)
  target_link_libraries(infer_aio PUBLIC infer_store Threads::Threads)
  target_compile_options(infer_aio PRIVATE -fvisibility=hidden)
  add_library(infer SHARED src/abi/abi.c)
  set_target_properties(infer PROPERTIES VERSION 3.0.0 SOVERSION 3)
  target_compile_options(infer PRIVATE -fvisibility=hidden)
  target_link_libraries(infer PRIVATE $<LINK_LIBRARY:WHOLE_ARCHIVE,infer_aio,infer_store,infer_io,infer_mem,infer_base>)
  target_link_options(infer PRIVATE -Wl,--exclude-libs,libinfer_storage.a:libinfer_memory.a)
  target_link_options(infer PRIVATE "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/abi/exports.map")
  set_property(TARGET infer APPEND PROPERTY LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/abi/exports.map")
endif()
if(INFER_BUILD_PRELOAD)
  add_library(infer_preload SHARED src/preload/preload.c)
  target_compile_options(infer_preload PRIVATE -fno-builtin-malloc -fno-builtin-calloc -fno-builtin-realloc -fno-builtin-free)
  target_link_libraries(infer_preload PRIVATE infer_mem ${CMAKE_DL_LIBS})
  target_link_options(infer_preload PRIVATE -Wl,--exclude-libs,ALL -Wl,-z,nodelete)
endif()
if(BUILD_TESTING)
  foreach(module base mem io)
    add_executable(test_${module} tests/test_${module}.c)
    target_link_libraries(test_${module} PRIVATE infer_${module} Threads::Threads)
    add_test(NAME ${module} COMMAND test_${module})
  endforeach()
  add_executable(test_io_backends tests/test_io_backends.c)
  target_link_libraries(test_io_backends PRIVATE infer_io Threads::Threads)
  if(INFER_IO_DLSYM)
    target_compile_definitions(test_io_backends PRIVATE IF_TEST_IO_DLSYM=1)
  endif()
  if(TARGET PkgConfig::V2_URING)
    target_compile_definitions(test_io_backends PRIVATE IF_TEST_IO_URING=1)
  endif()
  add_test(NAME io_backends COMMAND test_io_backends)
  add_test(NAME io_uring COMMAND test_io_backends uring)
  set_tests_properties(io_uring PROPERTIES SKIP_RETURN_CODE 77)
  if(INFER_BUILD_RUNTIME)
    foreach(module store aio runtime)
      add_executable(test_${module} tests/test_${module}.c)
      target_link_libraries(test_${module} PRIVATE infer_aio)
      add_test(NAME ${module} COMMAND test_${module})
    endforeach()
    if(INFER_ENABLE_CUDA)
      add_executable(test_runtime_gpu tests/test_runtime_gpu.cpp)
      target_link_libraries(test_runtime_gpu PRIVATE infer_store CUDA::cudart)
      add_test(NAME runtime_gpu COMMAND test_runtime_gpu)
    endif()
    add_executable(test_abi tests/test_abi.c)
    add_dependencies(test_abi infer)
    target_include_directories(test_abi PRIVATE include)
    target_link_libraries(test_abi PRIVATE ${CMAKE_DL_LIBS})
    add_test(NAME abi COMMAND test_abi $<TARGET_FILE:infer>)
  endif()
  if(INFER_BUILD_PRELOAD)
    add_executable(test_preload tests/test_preload.c)
    add_dependencies(test_preload infer_preload)
    target_include_directories(test_preload PRIVATE include)
    target_compile_options(test_preload PRIVATE -fno-builtin)
    target_link_libraries(test_preload PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
    if(NOT CMAKE_C_FLAGS MATCHES "fsanitize")
      add_test(NAME preload COMMAND ${CMAKE_COMMAND} -E env LD_PRELOAD=$<TARGET_FILE:infer_preload> $<TARGET_FILE:test_preload>)
    endif()
  endif()
endif()
if(INFER_BENCH AND INFER_BUILD_RUNTIME)
  add_executable(bench_store bench/bench_store.c)
  target_link_libraries(bench_store PRIVATE infer_aio)
endif()

# Keep C headers separate from the existing C++ infer/infer.h umbrella.
install(DIRECTORY include/infer DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/miniinfer)
install(TARGETS infer_base infer_mem infer_io ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
if(INFER_BUILD_RUNTIME)
  install(TARGETS infer LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
if(INFER_BUILD_PRELOAD)
  install(TARGETS infer_preload LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
